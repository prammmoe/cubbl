cmake_minimum_required(VERSION 3.22.1)
project(CubblTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(../../main/cpp/vendor/box2d box2d EXCLUDE_FROM_ALL)
set_target_properties(box2d PROPERTIES POSITION_INDEPENDENT_CODE ON)
enable_testing()
add_executable(core_tests CoreTests.cpp ../../main/cpp/game-core/Game.cpp ../../main/cpp/game-core/PhysicsWorld.cpp ../../main/cpp/procedural-scene/Building.cpp)
target_include_directories(core_tests PRIVATE ../../main/cpp)
target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME core_tests COMMAND core_tests)

add_executable(physics_tests PhysicsTests.cpp ../../main/cpp/game-core/Game.cpp ../../main/cpp/game-core/PhysicsWorld.cpp)
target_include_directories(physics_tests PRIVATE ../../main/cpp)
target_link_libraries(physics_tests PRIVATE box2d)
target_compile_options(physics_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME physics_tests COMMAND physics_tests)
target_link_libraries(core_tests PRIVATE box2d)
add_executable(game_integration_tests GameIntegrationTests.cpp ../../main/cpp/game-core/Game.cpp ../../main/cpp/game-core/PhysicsWorld.cpp)
target_include_directories(game_integration_tests PRIVATE ../../main/cpp)
target_link_libraries(game_integration_tests PRIVATE box2d)
target_compile_options(game_integration_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME game_integration_tests COMMAND game_integration_tests)

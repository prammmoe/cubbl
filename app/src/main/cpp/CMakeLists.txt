cmake_minimum_required(VERSION 3.22.1)
project(cubbl LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(vendor/box2d box2d EXCLUDE_FROM_ALL)
set_target_properties(box2d PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(cubbl SHARED main.cpp Renderer.cpp game-core/Game.cpp game-core/PhysicsWorld.cpp procedural-scene/Building.cpp)
find_package(game-activity REQUIRED CONFIG)
target_link_options(cubbl PRIVATE "-uJava_com_google_androidgamesdk_GameActivity_initializeNativeCode")
target_compile_options(cubbl PRIVATE -Wall -Wextra)
target_link_libraries(cubbl box2d game-activity::game-activity_static EGL GLESv3 android log)
